Visão Geral do Projeto

O Catálogo Ágil é o resultado de uma pesquisa profunda e engenharia reversa das principais plataformas do mercado, projetado para ser tecnicamente superior e comercialmente mais competitivo.

Problema Identificado

Plataformas atuais como "Vou Pedir" dependem de tecnologia white-label (iPaper), resultando em custos elevados (€595/mês), processos manuais de configuração e limitações de SEO e personalização.

💡

Nossa Solução

Plataforma 100% nativa com tecnologia proprietária, automação por IA para mapeamento de hotspots, SEO superior com SSR, e integrações nativas com PIX, WhatsApp e ERPs brasileiros.

@app.post("/api/v1/catalogs", response_model=CatalogResponse, tags=["Catalogs"])
async def create_catalog(
    name: str,
    pdf_file: UploadFile = File(...),
    excel_file: Optional[UploadFile] = File(None),
    background_tasks: BackgroundTasks = None,
    db: Session = Depends(get_db)
):
    """
    Create a new catalog from PDF upload.
    
    - **name**: Catalog name
    - **pdf_file**: PDF file containing the catalog pages
    - **excel_file**: Optional Excel file with product data
    """
    # Generate unique build ID
    build_id = str(uuid.uuid4())
    
    # Validate file types
    if not pdf_file.filename.endswith('.pdf'):
        raise HTTPException(status_code=400, detail="File must be a PDF")
    
    # Upload PDF to S3
    pdf_key = f"uploads/{build_id}/original.pdf"
    await s3_service.upload_file(pdf_file, pdf_key)
    
    # Upload Excel if provided
    excel_key = None
    if excel_file:
        if not excel_file.filename.endswith(('.xlsx', '.xls')):
            raise HTTPException(status_code=400, detail="Product file must be Excel format")
        excel_key = f"uploads/{build_id}/products.xlsx"
        await s3_service.upload_file(excel_file, excel_key)
    
    # Create catalog record
    db_catalog = catalog_models.Catalog(
        build_id=build_id,
        name=name,
        status="processing",
        pdf_url=pdf_key,
        excel_url=excel_key,
        created_at=datetime.utcnow()
    )
    db.add(db_catalog)
    db.commit()
    db.refresh(db_catalog)
    
    # Queue processing task
    background_tasks.add_task(
        processing_service.process_catalog,
        build_id=build_id,
        pdf_key=pdf_key,
        excel_key=excel_key
    )
    
    return CatalogResponse(
        build_id=build_id,
        name=name,
        status="processing",
        message="Catalog uploaded successfully. Processing started.",
        viewer_url=f"/catalog/{build_id}"
    )


@app.get("/api/v1/catalogs", response_model=List[CatalogListResponse], tags=["Catalogs"])
async def list_catalogs(
    skip: int = 0,
    limit: int = 20,
    db: Session = Depends(get_db)
):
    """List all catalogs with pagination"""
    catalogs = db.query(catalog_models.Catalog)\
        .order_by(catalog_models.Catalog.created_at.desc())\
        .offset(skip)\
        .limit(limit)\
        .all()
    return catalogs


@app.get("/api/v1/catalogs/{build_id}", response_model=CatalogResponse, tags=["Catalogs"])
async def get_catalog(build_id: str, db: Session = Depends(get_db)):
    """Get catalog details by build_id"""
    catalog = db.query(catalog_models.Catalog)\
        .filter(catalog_models.Catalog.build_id == build_id)\
        .first()
    
    if not catalog:
        raise HTTPException(status_code=404, detail="Catalog not found")
    
    return catalog


@app.get("/api/v1/catalogs/{build_id}/metadata", tags=["Catalogs"])
async def get_catalog_metadata(build_id: str, db: Session = Depends(get_db)):
    """
    Get full catalog metadata including pages, hotspots and products.
    This endpoint is used by the frontend viewer.
    """
    catalog = db.query(catalog_models.Catalog)\
        .filter(catalog_models.Catalog.build_id == build_id)\
        .first()
    
    if not catalog:
        raise HTTPException(status_code=404, detail="Catalog not found")
    
    # Get pages
    pages = db.query(catalog_models.Page)\
        .filter(catalog_models.Page.catalog_id == catalog.id)\
        .order_by(catalog_models.Page.page_number)\
        .all()
    
    # Get products
    products = db.query(catalog_models.Product)\
        .filter(catalog_models.Product.catalog_id == catalog.id)\
        .all()
    
    # Build metadata response
    return {
        "build_id": build_id,
        "name": catalog.name,
        "status": catalog.status,
        "total_pages": len(pages),
        "pages": [
            {
                "number": page.page_number,
                "width": page.width,
                "height": page.height,
                "tiles": {
                    "base_url": f"{settings.CDN_URL}/tiles/{build_id}/{page.page_number}",
                    "levels": page.tile_levels or 3
                },
                "hotspots": [
                    {
                        "id": hs.id,
                        "product_id": hs.product_id,
                        "x": hs.x,
                        "y": hs.y,
                        "width": hs.width,
                        "height": hs.height
                    }
                    for hs in page.hotspots
                ]
            }
            for page in pages
        ],
        "products": {
            str(p.id): {
                "sku": p.sku,
                "name": p.name,
                "price": float(p.price),
                "unit": p.unit,
                "description": p.description,
                "image_url": p.image_url
            }
            for p in products
        },
        "settings": {
            "checkout_whatsapp": catalog.whatsapp_number,
            "checkout_email": catalog.email,
            "checkout_webhook": catalog.webhook_url,
            "pix_enabled": catalog.pix_enabled,
            "company_name": catalog.company_name
        }
    }

@app.post("/api/v1/orders", response_model=OrderResponse, tags=["Orders"])
async def create_order(
    order: OrderCreate,
    background_tasks: BackgroundTasks,
    db: Session = Depends(get_db)
):
    """
    Create a new order from cart items.
    Supports multiple checkout methods: WhatsApp, Email, Webhook, PIX.
    """
    # Validate catalog exists
    catalog = db.query(catalog_models.Catalog)\
        .filter(catalog_models.Catalog.build_id == order.build_id)\
        .first()
    
    if not catalog:
        raise HTTPException(status_code=404, detail="Catalog not found")
    
    # Calculate total
    total = sum(item.quantity * item.unit_price for item in order.items)
    
    # Create order
    db_order = catalog_models.Order(
        order_id=str(uuid.uuid4()),
        catalog_id=catalog.id,
        customer_name=order.customer_name,
        customer_phone=order.customer_phone,
        customer_email=order.customer_email,
        items=order.items,
        total=total,
        status="pending",
        checkout_method=order.checkout_method,
        created_at=datetime.utcnow()
    )
    db.add(db_order)
    db.commit()
    db.refresh(db_order)
    
    # Process checkout based on method
    if order.checkout_method == "whatsapp":
        # Generate WhatsApp message
        message = processing_service.generate_whatsapp_message(db_order, catalog)
        whatsapp_url = f"https://api.whatsapp.com/send?phone={catalog.whatsapp_number}&text={message}"
        
        return OrderResponse(
            order_id=db_order.order_id,
            status="pending",
            total=total,
            checkout_url=whatsapp_url,
            message="Order created. Click the link to send via WhatsApp."
        )
    
    elif order.checkout_method == "webhook":
        # Send to webhook
        background_tasks.add_task(
            processing_service.send_webhook,
            webhook_url=catalog.webhook_url,
            order=db_order
        )
        
        return OrderResponse(
            order_id=db_order.order_id,
            status="sent",
            total=total,
            message="Order sent to ERP via webhook."
        )
    
    elif order.checkout_method == "pix":
        # Generate PIX code
        pix_data = processing_service.generate_pix(total, catalog)
        
        return OrderResponse(
            order_id=db_order.order_id,
            status="awaiting_payment",
            total=total,
            pix_code=pix_data["code"],
            pix_qrcode=pix_data["qrcode"],
            message="PIX payment code generated."
        )
    
    return OrderResponse(
        order_id=db_order.order_id,
        status="pending",
        total=total,
        message="Order created successfully."
    )

// Generate metadata for SEO
export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const catalog = await getCatalogMetadata(params.buildId);
  
  if (!catalog) {
    return { title: 'Catálogo não encontrado' };
  }
  
  return {
    title: `${catalog.name} | Catálogo Digital`,
    description: `Explore o catálogo digital interativo de ${catalog.settings.company_name}. ${catalog.total_pages} páginas de produtos.`,
    openGraph: {
      title: catalog.name,
      description: `Catálogo digital com ${catalog.total_pages} páginas`,
      type: 'website',
    },
    other: {
      'application/ld+json': JSON.stringify({
        '@context': 'https://schema.org',
        '@type': 'ItemList',
        'name': catalog.name,
        'numberOfItems': Object.keys(catalog.products).length,
        'itemListElement': Object.values(catalog.products).map((product: any, index: number) => ({
          '@type': 'ListItem',
          'position': index + 1,
          'item': {
            '@type': 'Product',
            'name': product.name,
            'sku': product.sku,
            'offers': {
              '@type': 'Offer',
              'price': product.price,
              'priceCurrency': 'BRL'
            }
          }
        }))
      })
    }
  };
}

export default async function CatalogPage({ params }: PageProps) {
  const catalog = await getCatalogMetadata(params.buildId);
  
  if (!catalog) {
    notFound();
  }
  
  return (
    <main className="min-h-screen bg-gray-100">
      <CatalogViewer 
        buildId={params.buildId}
        initialData={catalog}
      />
    </main>
  );
}

      setCart(JSON.parse(savedCart));
    }
  }, [buildId]);
  
  // Save cart to localStorage
  useEffect(() => {
    localStorage.setItem(`cart-${buildId}`, JSON.stringify(cart));
  }, [cart, buildId]);
  
  // Track page view
  useEffect(() => {
    fetch('/api/v1/analytics/event', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        build_id: buildId,
        event_type: 'page_view',
        event_data: { page: currentPage }
      })
    });
  }, [currentPage, buildId]);
  
  const goToPage = useCallback((page: number) => {
    if (page >= 0 && page < totalPages) {
      setCurrentPage(page);
    }
  }, [totalPages]);
  
  const addToCart = useCallback((productId: string, quantity: number = 1) => {
    const product = products[productId];
    if (!product) return;
    
    setCart(prev => {
      const existing = prev.find(item => item.productId === productId);
      if (existing) {
        return prev.map(item =>
          item.productId === productId
            ? { ...item, quantity: item.quantity + quantity }
            : item
        );
      }
      return [...prev, {
        productId,
        sku: product.sku,
        name: product.name,
        price: product.price,
        quantity
      }];
    });
    
    // Track event
    fetch('/api/v1/analytics/event', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        build_id: buildId,
        event_type: 'add_to_cart',
        event_data: { product_id: productId, quantity }
      })
    });
  }, [products, buildId]);
  
  const removeFromCart = useCallback((productId: string) => {
    setCart(prev => prev.filter(item => item.productId !== productId));
  }, []);
  
  const updateQuantity = useCallback((productId: string, quantity: number) => {
    if (quantity <= 0) {
      removeFromCart(productId);
      return;
    }
    setCart(prev =>
      prev.map(item =>
        item.productId === productId ? { ...item, quantity } : item
      )
    );
  }, [removeFromCart]);
  
  const currentPageData = pages[currentPage];
  
  return (
    <div className="flex flex-col h-screen bg-gray-900">
      {/* Header */}
      <header className="flex items-center justify-between px-4 py-3 bg-gray-800 text-white">
        <h1 className="text-lg font-semibold">{initialData.name}</h1>
        <div className="flex items-center gap-4">
          <span className="text-sm text-gray-400">
            Página {currentPage + 1} de {totalPages}
          </span>
          <button
            onClick={() => setIsCartOpen(true)}
            className="relative p-2 bg-blue-600 rounded-lg hover:bg-blue-700 transition"
          >
            🛒
            {cart.length > 0 && (
              <span className="absolute -top-1 -right-1 w-5 h-5 bg-red-500 rounded-full text-xs flex items-center justify-center">
                {cart.length}
              </span>
            )}
          </button>
        </div>
      </header>
      
      {/* Viewer */}
      <div 
        ref={containerRef}
        className="flex-1 relative overflow-auto"
      >
        <div 
          className="relative mx-auto"
          style={{
            width: currentPageData.width * zoom,
            height: currentPageData.height * zoom,
            transform: `scale(${zoom})`,
            transformOrigin: 'top center'
          }}
        >
          {/* Page Image */}
          <Image
            src={`${currentPageData.tiles.base_url}/full.webp`}
            alt={`Página ${currentPage + 1}`}
            width={currentPageData.width}
            height={currentPageData.height}
            priority
            className="w-full h-auto"
          />
          
          {/* Hotspots */}
          {currentPageData.hotspots.map(hotspot => (
            <Hotspot
              key={hotspot.id}
              hotspot={hotspot}
              product={products[hotspot.product_id]}
              zoom={zoom}
              onAddToCart={addToCart}
            />
          ))}
        </div>
      </div>
      
      {/* Navigation */}
      <nav className="flex items-center justify-center gap-4 px-4 py-3 bg-gray-800">
        <button
          onClick={() => goToPage(currentPage - 1)}
          disabled={currentPage === 0}
          className="px-4 py-2 bg-gray-700 text-white rounded-lg disabled:opacity-50 disabled:cursor-not-allowed hover:bg-gray-600 transition"
        >
          ← Anterior
        </button>
        
        <div className="flex items-center gap-2">
          <button onClick={() => setZoom(z => Math.max(0.5, z - 0.25))} className="p-2 bg-gray-700 rounded">
            −
          </button>
          <span className="text-white text-sm w-16 text-center">{Math.round(zoom * 100)}%</span>
          <button onClick={() => setZoom(z => Math.min(2, z + 0.25))} className="p-2 bg-gray-700 rounded">
            +
          </button>
        </div>
        
        <button
          onClick={() => goToPage(currentPage + 1)}
          disabled={currentPage === totalPages - 1}
          className="px-4 py-2 bg-gray-700 text-white rounded-lg disabled:opacity-50 disabled:cursor-not-allowed hover:bg-gray-600 transition"
        >
          Próxima →
        </button>
      </nav>
      
      {/* Cart Sidebar */}
      <Cart
        isOpen={isCartOpen}
        onClose={() => setIsCartOpen(false)}
        items={cart}
        onUpdateQuantity={updateQuantity}
        onRemove={removeFromCart}
        settings={settings}
        buildId={buildId}
      />
    </div>
  );
}

"""
Catálogo Ágil - PDF Processing Worker
Celery worker for asynchronous PDF processing and tile generation
"""
import os
import json
import subprocess
import tempfile
from pathlib import Path
from celery import Celery
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import boto3
from PIL import Image
import pyvips

# Configuration
REDIS_URL = os.getenv('REDIS_URL', 'redis://localhost:6379/0')
DATABASE_URL = os.getenv('DATABASE_URL')
S3_BUCKET = os.getenv('S3_BUCKET', 'catalogo-agil-assets')
AWS_REGION = os.getenv('AWS_REGION', 'sa-east-1')

# Celery app
app = Celery('worker', broker=REDIS_URL, backend=REDIS_URL)

# Database
engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(bind=engine)

# S3 client
s3 = boto3.client('s3', region_name=AWS_REGION)


@app.task(bind=True, max_retries=3)
def process_catalog(self, build_id: str, pdf_key: str, excel_key: str = None):
    """
    Main task to process a catalog PDF.
    
    Pipeline:
    1. Download PDF from S3
    2. Extract pages as images (pdftoppm)
    3. Generate multi-resolution tiles (libvips)
    4. Upload tiles to S3
    5. Parse Excel for product data (if provided)
    6. Detect hotspots using OCR/AI
    7. Update database with metadata
    """
    db = SessionLocal()
    
    try:
        print(f"[{build_id}] Starting catalog processing...")
        
        # Update status
        update_catalog_status(db, build_id, 'processing')
        
        with tempfile.TemporaryDirectory() as tmpdir:
            tmpdir = Path(tmpdir)
            
            # Step 1: Download PDF
            print(f"[{build_id}] Downloading PDF...")
            pdf_path = tmpdir / 'catalog.pdf'
            s3.download_file(S3_BUCKET, pdf_key, str(pdf_path))
            
            # Step 2: Extract pages
            print(f"[{build_id}] Extracting pages...")
            pages_dir = tmpdir / 'pages'
            pages_dir.mkdir()
            
            # Use pdftoppm for high-quality extraction
            subprocess.run([
                'pdftoppm',
                '-png',
                '-r', '300',  # 300 DPI
                str(pdf_path),
                str(pages_dir / 'page')
            ], check=True)
            
            # Get list of extracted pages
            page_files = sorted(pages_dir.glob('page-*.png'))
            total_pages = len(page_files)
            print(f"[{build_id}] Extracted {total_pages} pages")
            
            # Step 3: Generate tiles for each page
            tiles_dir = tmpdir / 'tiles'
            tiles_dir.mkdir()
            
            page_metadata = []
            
            for idx, page_file in enumerate(page_files):
                page_num = idx + 1
                print(f"[{build_id}] Processing page {page_num}/{total_pages}...")
                
                # Load image with pyvips for efficient processing
                img = pyvips.Image.new_from_file(str(page_file))
                width, height = img.width, img.height
                
                # Create page directory
                page_tiles_dir = tiles_dir / str(page_num)
                page_tiles_dir.mkdir()
                
                # Generate full resolution WebP
                full_path = page_tiles_dir / 'full.webp'
                img.webpsave(str(full_path), Q=85)
                
                # Generate thumbnail
                thumb = img.thumbnail_image(400)
                thumb_path = page_tiles_dir / 'thumb.webp'
                thumb.webpsave(str(thumb_path), Q=80)
                
                # Generate multi-resolution tiles (for zoom)
                tile_levels = 3
                for level in range(tile_levels):
                    scale = 1 / (2 ** level)
                    level_dir = page_tiles_dir / f'level_{level}'
                    level_dir.mkdir()
                    
                    scaled = img.resize(scale)
                    
                    # Generate 512x512 tiles
                    tile_size = 512
                    for y in range(0, scaled.height, tile_size):
                        for x in range(0, scaled.width, tile_size):
                            tile = scaled.crop(
                                x, y,
                                min(tile_size, scaled.width - x),
                                min(tile_size, scaled.height - y)
                            )
                            tile_path = level_dir / f'{x}_{y}.webp'
                            tile.webpsave(str(tile_path), Q=80)
                
                page_metadata.append({
                    'page_number': page_num,
                    'width': width,
                    'height': height,
                    'tile_levels': tile_levels
                })
                
                # Upload tiles to S3
                for file_path in page_tiles_dir.rglob('*'):
                    if file_path.is_file():
                        s3_key = f"tiles/{build_id}/{file_path.relative_to(tiles_dir)}"
                        s3.upload_file(
                            str(file_path),
                            S3_BUCKET,
                            s3_key,
                            ExtraArgs={
                                'ContentType': 'image/webp',
                                'CacheControl': 'public, max-age=31536000'
                            }
                        )
            
            # Step 4: Parse Excel (if provided)
            products = []
            if excel_key:
                print(f"[{build_id}] Parsing product data...")
                excel_path = tmpdir / 'products.xlsx'
                s3.download_file(S3_BUCKET, excel_key, str(excel_path))
                products = parse_excel(excel_path)
            
            # Step 5: Detect hotspots (simplified - would use AI in production)
            print(f"[{build_id}] Detecting hotspots...")
            hotspots = detect_hotspots(page_files, products)
            
            # Step 6: Save to database
            print(f"[{build_id}] Saving metadata...")
            save_catalog_metadata(db, build_id, page_metadata, products, hotspots)
            
            # Update status to ready
            update_catalog_status(db, build_id, 'ready')
            
            print(f"[{build_id}] Processing complete!")
            
            return {
                'build_id': build_id,
                'status': 'ready',
                'total_pages': total_pages,
                'total_products': len(products)
            }
            
    except Exception as e:
        print(f"[{build_id}] Error: {str(e)}")
        update_catalog_status(db, build_id, 'error', str(e))
        raise self.retry(exc=e, countdown=60)
    
    finally:
        db.close()


def parse_excel(excel_path: Path) -> list:
    """Parse product data from Excel file"""
    import pandas as pd
    
    df = pd.read_excel(excel_path)
    products = []
    
    for _, row in df.iterrows():
        products.append({
            'sku': str(row.get('SKU', row.get('Código', ''))),
            'name': str(row.get('Nome', row.get('Produto', ''))),
            'price': float(row.get('Preço', row.get('Valor', 0))),
            'unit': str(row.get('Unidade', 'UN')),
            'description': str(row.get('Descrição', ''))
        })
    
    return products

Guia de Deployment

Instruções completas para colocar o projeto em produção.

1

Pré-requisitos

# Instalar dependências
sudo apt update && sudo apt install -y docker.io docker-compose nodejs npm python3.11 python3-pip

# Instalar AWS CLI
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip && sudo ./aws/install

# Configurar AWS
aws configure

# Instalar Terraform
wget https://releases.hashicorp.com/terraform/1.6.0/terraform_1.6.0_linux_amd64.zip
unzip terraform_1.6.0_linux_amd64.zip && sudo mv terraform /usr/local/bin/
2

Clonar e Configurar

# Clonar repositório
git clone https://github.com/seu-usuario/catalogo-agil.git
cd catalogo-agil

# Copiar e editar variáveis de ambiente
cp .env.example .env
nano .env

# Conteúdo do .env:
DATABASE_URL=postgresql://postgres:SUA_SENHA@localhost:5432/catalogo_agil
REDIS_URL=redis://localhost:6379/0
AWS_ACCESS_KEY_ID=SUA_CHAVE
AWS_SECRET_ACCESS_KEY=SEU_SEGREDO
S3_BUCKET=catalogo-agil-assets
AWS_REGION=sa-east-1
3

Provisionar Infraestrutura

# Navegar para diretório de infra
cd infra/terraform

# Inicializar Terraform
terraform init

# Planejar mudanças
terraform plan -out=tfplan

# Aplicar infraestrutura
terraform apply tfplan

# Salvar outputs
terraform output > ../outputs.txt
4

Deploy Local (Desenvolvimento)

# Voltar para raiz do projeto
cd ../..

# Subir todos os serviços
docker-compose up --build -d

# Verificar status
docker-compose ps

# Ver logs
docker-compose logs -f backend

# Acessar:
# - Frontend: http://localhost:3000
# - API Docs: http://localhost:8000/docs
5

Deploy Produção (AWS)

# Build das imagens
docker build -t catalogo-backend ./backend
docker build -t catalogo-frontend ./frontend
docker build -t catalogo-worker ./worker

# Push para ECR
aws ecr get-login-password --region sa-east-1 | docker login --username AWS --password-stdin SEU_ACCOUNT.dkr.ecr.sa-east-1.amazonaws.com

docker tag catalogo-backend:latest SEU_ACCOUNT.dkr.ecr.sa-east-1.amazonaws.com/catalogo-backend:latest
docker push SEU_ACCOUNT.dkr.ecr.sa-east-1.amazonaws.com/catalogo-backend:latest

# Deploy via ECS (ou usar GitHub Actions para CI/CD)
aws ecs update-service --cluster catalogo-cluster --service backend --force-new-deployment
6

Verificar Deploy

# Testar health check
curl https://api.catalogoagil.com.br/health

# Testar upload de catálogo
curl -X POST https://api.catalogoagil.com.br/api/v1/catalogs \
  -F "name=Teste" \
  -F "pdf_file=@catalogo-teste.pdf"

# Verificar logs
aws logs tail /ecs/catalogo-backend --follow
